import React, { useEffect, useRef, useState } from 'react'; import { Loader2 } from 'lucide-react'; const PDFFlipbook = () => { const flipbookRef = useRef(null); const [loading, setLoading] = useState(true); const [error, setError] = useState(null); useEffect(() => { const loadPDF = async () => { try { const pdfjsLib = window.pdfjsLib; if (!pdfjsLib) { throw new Error('PDF.js library not loaded'); } // Configure worker pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js'; // Load the PDF file using window.fs.readFile const pdfData = await window.fs.readFile('IND211_Portfolio.pdf'); const pdf = await pdfjsLib.getDocument({ data: pdfData }).promise; const totalPages = pdf.numPages; // Create front cover const frontCover = document.createElement('div'); frontCover.className = 'bg-gray-800 text-white flex items-center justify-center'; frontCover.innerHTML = 'Front Cover'; flipbookRef.current.appendChild(frontCover); // Render PDF pages for (let pageNum = 1; pageNum <= totalPages; pageNum++) { const page = await pdf.getPage(pageNum); const scale = 1.5; const viewport = page.getViewport({ scale }); const canvas = document.createElement('canvas'); const context = canvas.getContext('2d'); canvas.width = viewport.width; canvas.height = viewport.height; await page.render({ canvasContext: context, viewport: viewport }).promise; const dataUrl = canvas.toDataURL('image/jpeg', 0.8); const pageDiv = document.createElement('div'); pageDiv.className = 'page bg-white'; pageDiv.style.backgroundImage = `url(${dataUrl})`; pageDiv.style.backgroundSize = 'contain'; pageDiv.style.backgroundRepeat = 'no-repeat'; pageDiv.style.backgroundPosition = 'center'; flipbookRef.current.appendChild(pageDiv); } // Create back cover const backCover = document.createElement('div'); backCover.className = 'bg-gray-800 text-white flex items-center justify-center'; backCover.innerHTML = 'Back Cover'; flipbookRef.current.appendChild(backCover); // Initialize Turn.js window.jQuery(flipbookRef.current).turn({ width: 800, height: 600, autoCenter: true, elevation: 50, gradients: true, display: 'double', acceleration: true, when: { turned: function(e, page) { console.log('Current page:', page); } } }); setLoading(false); } catch (err) { console.error('Error loading PDF:', err); setError(err.message); setLoading(false); } }; // Load required scripts const loadScripts = async () => { try { // Load jQuery const jqueryScript = document.createElement('script'); jqueryScript.src = 'https://cdnjs.cloudflare.com/ajax/libs/jquery/3.6.0/jquery.min.js'; await new Promise((resolve, reject) => { jqueryScript.onload = resolve; jqueryScript.onerror = reject; document.head.appendChild(jqueryScript); }); // Load Turn.js const turnScript = document.createElement('script'); turnScript.src = 'https://cdnjs.cloudflare.com/ajax/libs/turn.js/4.1.0/turn.min.js'; await new Promise((resolve, reject) => { turnScript.onload = resolve; turnScript.onerror = reject; document.head.appendChild(turnScript); }); // Load PDF.js const pdfScript = document.createElement('script'); pdfScript.src = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js'; await new Promise((resolve, reject) => { pdfScript.onload = resolve; pdfScript.onerror = reject; document.head.appendChild(pdfScript); }); // Now that scripts are loaded, initialize the PDF await loadPDF(); } catch (err) { console.error('Error loading scripts:', err); setError('Failed to load required libraries'); setLoading(false); } }; loadScripts(); // Cleanup return () => { if (flipbookRef.current) { window.jQuery(flipbookRef.current).turn('destroy'); } }; }, []); if (error) { return (
Error: {error}
); } return (
{loading ? (
) : (
)}
); }; export default PDFFlipbook;
